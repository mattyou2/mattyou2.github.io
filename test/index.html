// We importeren de AI-bibliotheek import { pipeline } from 'https://jsdelivr.net'; let aiModel = null; const statusHeader = document.getElementById('status'); const inputField = document.getElementById('user-input'); const sendBtn = document.getElementById('send-btn'); const chatBox = document.getElementById('chat-box'); // Functie die de AI op de achtergrond inlaadt async function loadModel() { try { if (!aiModel) { // We laden het model aiModel = await pipeline('text2text-generation', 'Xenova/LaMini-Flan-T5-78M'); } return aiModel; } catch (err) { console.error("Model laadfout:", err); return null; } } async function handleChat() { const text = inputField.value.trim(); if (!text) return; // Doe niks als de balk leeg is // Toon direct jouw bericht op het scherm appendMessage(text, 'user'); inputField.value = ''; // Maak de typbalk meteen weer leeg // Toon dat de AI bezig is statusHeader.innerText = "AI denkt na..."; statusHeader.style.background = "#ff9800"; try { // Laad het model (of pak de al geladen versie) const model = await loadModel(); if (model) { // Genereer het antwoord const response = await model(text, { max_new_tokens: 60, temperature: 0.7 }); // HIER ZAT DE FOUT: we halen de tekst nu op de juiste manier op const aiText = response[0]?.generated_text || response.generated_text; if (aiText) { appendMessage(aiText, 'ai'); } else { appendMessage("Ik kon geen logisch antwoord genereren.", 'ai'); } } else { appendMessage("Oeps, ik kon het AI-model niet lokaal opstarten.", 'ai'); } } catch (error) { appendMessage("Er is een fout opgetreden bij het berekenen van het antwoord.", 'ai'); console.error(error); } // Zet status weer terug naar normaal statusHeader.innerText = "Mijn Lokale AI Assistent"; statusHeader.style.background = "#007bff"; } function appendMessage(text, sender) { const div = document.createElement('div'); div.classList.add('msg', sender); div.innerText = text; chatBox.appendChild(div); chatBox.scrollTop = chatBox.scrollHeight; // Scroll automatisch omlaag } // Luister naar de klik op de knop en de Enter-toets sendBtn.addEventListener('click', handleChat); inputField.addEventListener('keypress', (e) => { if (e.key === 'Enter') handleChat(); });
