# -------------------------------------------------------------------
# MattyouSMP Economy System
# -------------------------------------------------------------------

# Ensures new players start with €0 (change 0 to any starting amount)
on join:
    if {money::%player's uuid%} is not set:
        set {money::%player's uuid%} to 0

# The Actionbar at the bottom of the screen (updates every second)
every 1 second:
    loop all players:
        set {_m} to {money::%loop-player's uuid%} if {money::%loop-player's uuid%} is set else 0
        send action bar "&6&lMattyouSMP Money: &e€%{_m}%" to loop-player

# -------------------------------------------------------------------
# Player Command: /pay
# -------------------------------------------------------------------
command /pay [<player>] [<number>]:
    trigger:
        if arg 1 is not set:
            send "&cUsage: /pay <player> <amount>" to player
            stop
        if arg 2 is not set:
            send "&cUsage: /pay <player> <amount>" to player
            stop
        if arg 2 <= 0:
            send "&cAmount must be greater than 0!" to player
            stop
        
        # Check if the player has enough money
        if {money::%player's uuid%} < arg 2:
            send "&cYou don't have enough money! You currently have &e€%{money::%player's uuid%}%&c." to player
            stop
            
        # Execute the transaction
        subtract arg 2 from {money::%player's uuid%}
        add arg 2 to {money::%arg 1's uuid%}
        
        send "&aYou sent &e€%arg 2% &ato &7%arg 1%&a." to player
        send "&aYou received &e€%arg 2% &afrom &7%player%&a." to arg 1

# -------------------------------------------------------------------
# Admin Command: /money (set/add/remove)
# -------------------------------------------------------------------
command /money [<text>] [<player>] [<number>]:
    permission: op
    permission message: &cYou do not have permission to execute this command!
    trigger:
        if arg 1 is not set:
            send "&cUsage: /money (set/add/remove) <player> <amount>" to player
            stop
            
        # If an admin types just /money <player> to check balance
        if arg 1 is set:
            if arg 1 is not "set" or "add" or "remove":
                # Display balance if the first argument is a valid player name
                set {_p} to arg 1 parsed as player
                if {_p} is set:
                    send "&eBalance of %{_p}%: &a€%{money::%{_p}'s uuid%}%" to player
                    stop
                else:
                    send "&cUsage: /money (set/add/remove) <player> <amount>" to player
                    stop

        if arg 2 is not set:
            send "&cPlease specify a player." to player
            stop
        if arg 3 is not set:
            send "&cPlease specify an amount." to player
            stop

        # Admin Actions
        if arg 1 is "set":
            set {money::%arg 2's uuid%} to arg 3
            send "&aBalance of &7%arg 2% &aset to &e€%arg 3%&a." to player
            send "&aAn admin has set your balance to &e€%arg 3%&a." to arg 2

        else if arg 1 is "add":
            add arg 3 to {money::%arg 2's uuid%}
            send "&e€%arg 3% &ahass been added to &7%arg 2%&a." to player
            send "&a&e€%arg 3% &ahas been added to your balance." to arg 2

        else if arg 1 is "remove":
            subtract arg 3 from {money::%arg 2's uuid%}
            send "&e€%arg 3% &chas been deducted from &7%arg 2%&a." to player
            send "&c&e€%arg 3% &chas been deducted from your balance." to arg 2
